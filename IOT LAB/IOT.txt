#include <Wire.h>
#include <RTClib.h>
#include <LiquidCrystal_I2C.h>

RTC_DS3231 rtc;
LiquidCrystal_I2C lcd(0x27, 16, 2);

// I2C pins
#define SDA_PIN 21
#define SCL_PIN 22

// LED pins
#define LED_MORNING   13
#define LED_AFTERNOON 12
#define LED_NIGHT     14

// Buzzer
#define BUZZER 25

void setup() {
  Serial.begin(115200);

  Wire.begin(SDA_PIN, SCL_PIN);

  lcd.init();
  lcd.backlight();

  if (!rtc.begin()) {
    lcd.print("RTC ERROR");
    while (1);
  }

  // ✅ TIME SET (AUTO FROM PC)
// rtc.adjust(DateTime(2026, 5, 18, 14, 35, 0));

  pinMode(LED_MORNING, OUTPUT);
  pinMode(LED_AFTERNOON, OUTPUT);
  pinMode(LED_NIGHT, OUTPUT);
  pinMode(BUZZER, OUTPUT);

  allOff();

  lcd.setCursor(0, 0);
  lcd.print("Medicine Box");
  lcd.setCursor(0, 1);
  lcd.print("Starting...");
  delay(2000);
  lcd.clear();
}

void loop() {
  DateTime now = rtc.now();

  int h = now.hour();
  int m = now.minute();

  lcd.setCursor(0, 0);
  lcd.print("Time ");
  lcd.print(h);
  lcd.print(":");
  if (m < 10) lcd.print("0");
  lcd.print(m);
  lcd.print("   ");

  if (h == 15 && m == 48) {
    runAlarm("GOOD MORNING", "TAKE MEDICINE", LED_MORNING);
  }
  else if (h == 15 && m == 25 ) {
    runAlarm("GOOD AFTERNOON", "TAKE MEDICINE", LED_AFTERNOON);
  }
  else if (h == 15&& m == 26) {
    runAlarm("GOOD NIGHT", "TAKE MEDICINE", LED_NIGHT);
  }

  delay(1000);
}

// -------- ALARM FUNCTION --------
void runAlarm(String line1, String line2, int ledPin) {

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print(line1);
  lcd.setCursor(0, 1);
  lcd.print(line2);

  for (int i = 0; i < 3; i++) {  // ~20 sec

    digitalWrite(ledPin, HIGH);

    digitalWrite(BUZZER, HIGH);
    delay(5000);

    digitalWrite(BUZZER, LOW);
    delay(2000);

    digitalWrite(BUZZER, HIGH);
    delay(5000);

    digitalWrite(BUZZER, LOW);
    delay(2000);
  }

  allOff();
  lcd.clear();
}

// -------- ALL OFF --------
void allOff() {
  digitalWrite(LED_MORNING, LOW);
  digitalWrite(LED_AFTERNOON, LOW);
  digitalWrite(LED_NIGHT, LOW);
  digitalWrite(BUZZER, LOW);
}
